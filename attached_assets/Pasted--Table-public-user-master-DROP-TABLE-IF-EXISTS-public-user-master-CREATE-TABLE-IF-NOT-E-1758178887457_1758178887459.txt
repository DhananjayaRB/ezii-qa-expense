-- Table: public.user_master

-- DROP TABLE IF EXISTS public.user_master;

CREATE TABLE IF NOT EXISTS public.user_master
(
    user_id bigint NOT NULL DEFAULT nextval('user_master_new_user_id_seq'::regclass),
    org_id bigint,
    employee_number character varying(50) COLLATE pg_catalog."default" NOT NULL,
    first_name character varying(100) COLLATE pg_catalog."default",
    middle_name character varying(100) COLLATE pg_catalog."default",
    last_name character varying(100) COLLATE pg_catalog."default",
    date_of_joining date,
    group_date_of_joining date,
    gender numeric(1,0),
    email character varying(150) COLLATE pg_catalog."default",
    communication_email character varying(150) COLLATE pg_catalog."default",
    date_of_birth date,
    user_type_ids integer[],
    role_id smallint,
    contacts jsonb,
    worker_type_id bigint,
    user_profile text COLLATE pg_catalog."default",
    other_roles bigint[],
    is_differently_abled numeric(1,0) DEFAULT 0,
    is_international_worker numeric(1,0) DEFAULT 0,
    is_pension_applicable numeric(1,0) DEFAULT 0,
    is_admin numeric(1,0),
    is_migrated numeric(1,0) DEFAULT 0,
    is_active numeric(1,0) DEFAULT 0,
    created_at timestamp with time zone DEFAULT now(),
    created_by bigint NOT NULL,
    updated_at timestamp with time zone DEFAULT now(),
    updated_by bigint,
    is_exited numeric(1,0) DEFAULT 0,
    is_rejoinee numeric(1,0) NOT NULL DEFAULT '0'::numeric,
    pre_user_id bigint DEFAULT '0'::bigint,
    is_rejoining numeric(1,0),
    core_id smallint,
    alternative_id character varying(100) COLLATE pg_catalog."default",
    biometric_id character varying(100) COLLATE pg_catalog."default",
    prev_user_id bigint,
    leave_id character varying(50) COLLATE pg_catalog."default" DEFAULT '0'::character varying,
	expense_id  character varying(50) COLLATE pg_catalog."default" DEFAULT '0'::character varying,
    attendance_id smallint,
    CONSTRAINT user_master_new_pkey PRIMARY KEY (user_id)
)

WITH (
    autovacuum_enabled = TRUE
)
TABLESPACE pg_default;
