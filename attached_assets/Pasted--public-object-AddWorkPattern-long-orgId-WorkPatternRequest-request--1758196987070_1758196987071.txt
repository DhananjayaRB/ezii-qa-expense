        public object AddWorkPattern(long orgId, WorkPatternRequest request)
        {
            const string insertQuery = @"
    INSERT INTO work_pattern (
        org_id, pattern_name, start_date, end_date,
        number_of_working_days, number_of_non_working_days,
        total_holidays, selected_holidays, work_type, 
        non_working_days, max_holidays_for_emp, custom_days,floatingConfig
    )
    VALUES (
        @OrgId, @PatternName, @StartDate, @EndDate,
        @NumberOfWorkingDays, @NumberOfNonWorkingDays,
        @TotalHolidays, @SelectedHolidays::int[], @WorkType, 
        @NonWorkingDays::jsonb, @MaxHolidaysForEmp, @CustomDays::jsonb,@total:: jsonb
    );
    ";

            int rowsAffected = ConnectionManager.GetConnection().Execute(insertQuery, new
            {
                OrgId = orgId,
                request.PatternName,
                request.StartDate,
                request.EndDate,
                request.NumberOfWorkingDays,
                request.NumberOfNonWorkingDays,
                request.TotalHolidays,
                SelectedHolidays = request.SelectedHolidays?.ToArray(),
                request.WorkType,
                NonWorkingDays = JsonConvert.SerializeObject(request.NonWorkingDays),
                CustomDays = JsonConvert.SerializeObject(request.CustomDays),
                request.MaxHolidaysForEmp,
                total = JsonConvert.SerializeObject(request.floatingConfig)
            });

            if (rowsAffected == 0)
                throw new Exception("Failed to add work pattern.");

            return new
            {
                request.PatternName,
                request.WorkType,
                request.StartDate,
                request.EndDate,
                request.NumberOfWorkingDays,
                request.NumberOfNonWorkingDays,
                request.TotalHolidays,
                request.SelectedHolidays,
                request.NonWorkingDays,
                request.CustomDays,
                request.MaxHolidaysForEmp,
                request.floatingConfig.total
            };
        }

        public List<object> GetWorkPatterns(long orgId)
        {
            var query = @"
SELECT 
    id,
    pattern_name,
    start_date,
    end_date,
    number_of_working_days,
    number_of_non_working_days,
    total_holidays,
    selected_holidays,
    work_type,
    non_working_days,
    max_holidays_for_emp,
    custom_days,employees,floatingConfig
FROM work_pattern
WHERE org_id = @OrgId
ORDER BY id DESC;
";

            var rows = ConnectionManager.GetConnection().Query(query, new { OrgId = orgId }).ToList();

            var result = rows.Select(row => new
            {
                id = row.id,
                patternName = row.pattern_name,
                startDate = row.start_date,
                endDate = row.end_date,
                numberOfWorkingDays = row.number_of_working_days,
                numberOfNonWorkingDays = row.number_of_non_working_days,
                totalHolidays = row.total_holidays,
                selectedHolidays = ((int[])row.selected_holidays).ToList(),
                workType = row.work_type,
                nonWorkingDays = JsonConvert.DeserializeObject<List<NonWorkingDayRequest>>(row.non_working_days.ToString()),
                maxHolidaysForEmp = row.max_holidays_for_emp,
                customDays = JsonConvert.DeserializeObject<List<CustomDayRequest>>(row.custom_days?.ToString() ?? "[]"),// NEW,
                employees = row.employees,
                floatingConfig = row.floatingconfig,
            }).ToList<object>();

            return result;
        }


        public object UpdateWorkPattern(long orgId, WorkPatternRequest request)
        {
            try
            {
                var parameters = new DynamicParameters();
                parameters.Add("OrgId", orgId);
                parameters.Add("Id", request.Id);
                parameters.Add("PatternName", request.PatternName);
                parameters.Add("StartDate", request.StartDate);
                parameters.Add("EndDate", request.EndDate);
                parameters.Add("NumberOfWorkingDays", request.NumberOfWorkingDays);
                parameters.Add("NumberOfNonWorkingDays", request.NumberOfNonWorkingDays);
                parameters.Add("TotalHolidays", request.TotalHolidays);
                parameters.Add("SelectedHolidays", request.SelectedHolidays?.ToArray());
                parameters.Add("WorkType", request.WorkType);
                parameters.Add("NonWorkingDays", JsonConvert.SerializeObject(request.NonWorkingDays));
                parameters.Add("CustomDays", JsonConvert.SerializeObject(request.CustomDays));
                parameters.Add("MaxHolidaysForEmp", request.MaxHolidaysForEmp);
                parameters.Add("total", request.floatingConfig);

                string query = @"
UPDATE work_pattern SET
    pattern_name = @PatternName,
    start_date = @StartDate,
    end_date = @EndDate,
    number_of_working_days = @NumberOfWorkingDays,
    number_of_non_working_days = @NumberOfNonWorkingDays,
    total_holidays = @TotalHolidays,
    selected_holidays = @SelectedHolidays::int[],
    work_type = @WorkType,
    non_working_days = @NonWorkingDays::jsonb,
    max_holidays_for_emp = @MaxHolidaysForEmp,total = @total :: jsonb,
    custom_days = @CustomDays::jsonb";

                // Conditionally add the `employees` update only if not null or empty
                if (request.employees != null) //&& request.employees.Any()
                {
                    query += ", employees = @Employees::bigint[]";
                    parameters.Add("Employees", request.employees.ToArray());
                }

                query += " WHERE id = @Id AND org_id = @OrgId;";

                int rows = ConnectionManager.GetConnection().Execute(query, parameters);

                if (rows == 0)
                    throw new Exception("No record updated. Invalid ID or OrgId.");

                return new
                {
                    request.Id,
                    request.PatternName,
                    request.WorkType,
                    request.StartDate,
                    request.EndDate,
                    request.NumberOfWorkingDays,
                    request.NumberOfNonWorkingDays,
                    request.TotalHolidays,
                    request.SelectedHolidays,
                    request.NonWorkingDays,
                    request.CustomDays,
                    request.MaxHolidaysForEmp,
                    request.floatingConfig.total
                };
            }
            catch (Exception ex)
            {
                throw new Exception(ex.Message);
            }
        }
